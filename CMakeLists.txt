cmake_minimum_required(VERSION 3.24)
project(PRESAGeCPU LANGUAGES CXX)
include(CTest)
find_program(PRESAGE_JULIA julia REQUIRED)
set(PRESAGE_PLAN "" CACHE FILEPATH "Existing evaluator plan; empty generates a fresh erf candidate")
set(PRESAGE_EVALUATION separate CACHE STRING "Horner arithmetic: separate or fma")
set(PRESAGE_INTERNAL fixed CACHE STRING "Internal arithmetic: fixed, single or double")
set(PRESAGE_BACKEND ac_fixed CACHE STRING "C++ arithmetic: portable, ap_fixed or ac_fixed")
set_property(CACHE PRESAGE_BACKEND PROPERTY STRINGS portable ap_fixed ac_fixed)
set(PRESAGE_FLOAT_BACKEND auto CACHE STRING "Floating arithmetic: auto, native or ac_std_float")
set_property(CACHE PRESAGE_FLOAT_BACKEND PROPERTY STRINGS auto native ac_std_float)
if(NOT PRESAGE_FLOAT_BACKEND MATCHES "^(auto|native|ac_std_float)$")
  message(FATAL_ERROR "PRESAGE_FLOAT_BACKEND must be auto, native or ac_std_float")
endif()
if(NOT PRESAGE_BACKEND MATCHES "^(portable|ap_fixed|ac_fixed)$")
  message(FATAL_ERROR "PRESAGE_BACKEND must be portable, ap_fixed or ac_fixed")
endif()
set(default_certify OFF)
if(PRESAGE_INTERNAL STREQUAL "fixed")
  set(default_certify ON)
endif()
option(PRESAGE_CERTIFY "Require a uniform evaluator certificate" ${default_certify})
file(GLOB PRESAGE_JULIA_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.jl" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp")
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(needs_gmp OFF)
set(needs_ac OFF)
if(PRESAGE_INTERNAL STREQUAL "fixed")
  set(needs_gmp ON)
endif()
if(PRESAGE_FLOAT_BACKEND STREQUAL "ac_std_float" OR PRESAGE_BACKEND STREQUAL "ac_fixed")
  set(needs_ac ON)
endif()
if(PRESAGE_PLAN)
  set(source "${PRESAGE_PLAN}")
  set(options)
  execute_process(
    COMMAND "${PRESAGE_JULIA}" --startup-file=no "--project=${CMAKE_CURRENT_SOURCE_DIR}"
      -e "using PRESAGe; p=read_plan(ARGS[1]).program; p===nothing && error(\"plan has no evaluator\"); print(PRESAGe._has_fixed(p) ? \"ON\" : \"OFF\", \" \", PRESAGe._resolved_float_backend(p,Symbol(ARGS[2]))==:ac_std_float ? \"ON\" : \"OFF\")"
      "${PRESAGE_PLAN}" "${PRESAGE_FLOAT_BACKEND}"
    RESULT_VARIABLE inspect_result OUTPUT_VARIABLE resources OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT inspect_result EQUAL 0)
    message(FATAL_ERROR "Could not inspect evaluator plan dependencies")
  endif()
  separate_arguments(resources UNIX_COMMAND "${resources}")
  list(GET resources 0 needs_gmp)
  list(GET resources 1 floating_needs_ac)
  if(floating_needs_ac)
    set(needs_ac ON)
  endif()
else()
  set(source erf)
  set(options "--evaluation=${PRESAGE_EVALUATION}" "--internal=${PRESAGE_INTERNAL}")
endif()
list(APPEND options "--backend=${PRESAGE_BACKEND}")
list(APPEND options "--float-backend=${PRESAGE_FLOAT_BACKEND}")
set(certificate_output)
if(PRESAGE_CERTIFY)
  list(APPEND options --certify=true)
  set(certificate_output "${generated}/certificate.toml")
else()
  list(APPEND options --certify=false)
endif()
add_custom_command(
  OUTPUT "${generated}/evaluator.hpp" "${generated}/evaluate.cpp" "${generated}/plan.toml" ${certificate_output}
  COMMAND "${PRESAGE_JULIA}" --startup-file=no "--project=${CMAKE_CURRENT_SOURCE_DIR}"
          "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate.jl" "${source}" "${generated}" ${options}
  DEPENDS ${PRESAGE_JULIA_SOURCES} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate.jl" ${PRESAGE_PLAN}
  VERBATIM)
add_executable(presage_cpu "${generated}/evaluate.cpp" "${generated}/evaluator.hpp")
target_compile_features(presage_cpu PRIVATE cxx_std_20)
if(needs_gmp)
  find_path(PRESAGE_GMP_INCLUDE_DIR gmpxx.h HINTS /opt/homebrew/opt/gmp/include REQUIRED)
  find_library(PRESAGE_GMPXX gmpxx HINTS /opt/homebrew/opt/gmp/lib REQUIRED)
  find_library(PRESAGE_GMP gmp HINTS /opt/homebrew/opt/gmp/lib REQUIRED)
  target_include_directories(presage_cpu SYSTEM PRIVATE "${PRESAGE_GMP_INCLUDE_DIR}")
  target_link_libraries(presage_cpu PRIVATE "${PRESAGE_GMPXX}" "${PRESAGE_GMP}")
endif()
if(PRESAGE_BACKEND STREQUAL "ap_fixed")
  find_path(PRESAGE_AP_INCLUDE_DIR ap_fixed.h
    HINTS "$ENV{PRESAGE_AP_INCLUDE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../HLS_arbitrary_Precision_Types/include"
    REQUIRED)
  target_include_directories(presage_cpu SYSTEM PRIVATE "${PRESAGE_AP_INCLUDE_DIR}")
endif()
find_path(PRESAGE_AC_INCLUDE_DIR ac_std_float.h
    HINTS "$ENV{PRESAGE_AC_INCLUDE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../ac_types/include"
    )
if(PRESAGE_AC_INCLUDE_DIR)
  target_include_directories(presage_cpu SYSTEM PRIVATE "${PRESAGE_AC_INCLUDE_DIR}")
elseif(needs_ac)
  message(FATAL_ERROR "AC headers required; set PRESAGE_AC_INCLUDE_DIR")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(presage_cpu PRIVATE -ffp-contract=off -fno-fast-math -frounding-math)
else()
  message(FATAL_ERROR "The CPU verification build currently requires Clang or GCC")
endif()
if(BUILD_TESTING)
  add_test(NAME cpu_conformance COMMAND "${PRESAGE_JULIA}" --startup-file=no
    "--project=${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_cpu.jl"
    "${generated}/plan.toml" "$<TARGET_FILE:presage_cpu>")
endif()
