cmake_minimum_required(VERSION 3.24)
project(PRESAGeCPU LANGUAGES CXX)
include(CTest)
option(PRESAGE_CHECKER_ONLY "Build the standalone checker without Julia or an evaluator" OFF)
option(PRESAGE_TRACE "Generate intermediate-value tracing for local debugging" OFF)
find_path(PRESAGE_GMP_INCLUDE_DIR gmpxx.h HINTS /opt/homebrew/opt/gmp/include REQUIRED)
find_library(PRESAGE_GMPXX gmpxx HINTS /opt/homebrew/opt/gmp/lib REQUIRED)
find_library(PRESAGE_GMP gmp HINTS /opt/homebrew/opt/gmp/lib REQUIRED)
find_path(PRESAGE_MPFR_INCLUDE_DIR mpfr.h HINTS /opt/homebrew/opt/mpfr/include REQUIRED)
find_library(PRESAGE_MPFR mpfr HINTS /opt/homebrew/opt/mpfr/lib REQUIRED)
add_executable(presage_check tools/verification/check.cpp)
target_compile_features(presage_check PRIVATE cxx_std_20)
target_include_directories(presage_check PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_include_directories(presage_check SYSTEM PRIVATE "${PRESAGE_GMP_INCLUDE_DIR}" "${PRESAGE_MPFR_INCLUDE_DIR}")
target_link_libraries(presage_check PRIVATE "${PRESAGE_MPFR}" "${PRESAGE_GMPXX}" "${PRESAGE_GMP}")
if(PRESAGE_CHECKER_ONLY)
  return()
endif()
find_program(PRESAGE_JULIA julia REQUIRED)
set(PRESAGE_PLAN "" CACHE FILEPATH "Existing evaluator plan; empty generates a fresh erf candidate")
set(PRESAGE_RESOURCE_MODEL "" CACHE FILEPATH "Optional editable resource profile and kernel memory policy")
set(PRESAGE_TARGET cpu CACHE STRING "Evaluator target: cpu or kernel")
set_property(CACHE PRESAGE_TARGET PROPERTY STRINGS cpu kernel)
set(PRESAGE_TOOLCHAIN generic CACHE STRING "Kernel memory hints: generic, vitis or oneapi")
set_property(CACHE PRESAGE_TOOLCHAIN PROPERTY STRINGS generic vitis oneapi)
if(NOT PRESAGE_TARGET MATCHES "^(cpu|kernel)$")
  message(FATAL_ERROR "PRESAGE_TARGET must be cpu or kernel")
endif()
if(NOT PRESAGE_TOOLCHAIN MATCHES "^(generic|vitis|oneapi)$")
  message(FATAL_ERROR "PRESAGE_TOOLCHAIN must be generic, vitis or oneapi")
endif()
if(PRESAGE_TARGET STREQUAL "kernel" AND NOT PRESAGE_PLAN)
  message(FATAL_ERROR "Kernel generation requires PRESAGE_PLAN with fixed arithmetic or encoded tables")
endif()
set(PRESAGE_EVALUATION separate CACHE STRING "Horner arithmetic: separate or fma")
set(PRESAGE_INTERNAL fixed CACHE STRING "Internal arithmetic: fixed, single or double")
set(PRESAGE_BACKEND ac_fixed CACHE STRING "C++ arithmetic: portable, ap_fixed or ac_fixed")
set_property(CACHE PRESAGE_BACKEND PROPERTY STRINGS portable ap_fixed ac_fixed)
set(PRESAGE_FLOAT_BACKEND auto CACHE STRING "Floating arithmetic: auto, native or ac_std_float")
set_property(CACHE PRESAGE_FLOAT_BACKEND PROPERTY STRINGS auto native ac_std_float)
if(NOT PRESAGE_FLOAT_BACKEND MATCHES "^(auto|native|ac_std_float)$")
  message(FATAL_ERROR "PRESAGE_FLOAT_BACKEND must be auto, native or ac_std_float")
endif()
if(NOT PRESAGE_BACKEND MATCHES "^(portable|ap_fixed|ac_fixed)$")
  message(FATAL_ERROR "PRESAGE_BACKEND must be portable, ap_fixed or ac_fixed")
endif()
set(default_certify OFF)
if(PRESAGE_INTERNAL STREQUAL "fixed")
  set(default_certify ON)
endif()
option(PRESAGE_CERTIFY "Require a uniform evaluator certificate" ${default_certify})
file(GLOB PRESAGE_JULIA_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.jl" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp")
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(needs_gmp OFF)
set(needs_ac OFF)
if(PRESAGE_INTERNAL STREQUAL "fixed")
  set(needs_gmp ON)
endif()
if(PRESAGE_FLOAT_BACKEND STREQUAL "ac_std_float" OR PRESAGE_BACKEND STREQUAL "ac_fixed")
  set(needs_ac ON)
endif()
if(PRESAGE_PLAN)
  set(source "${PRESAGE_PLAN}")
  set(options)
  execute_process(
    COMMAND "${PRESAGE_JULIA}" --startup-file=no "--project=${CMAKE_CURRENT_SOURCE_DIR}"
      -e [=[
        using PRESAGe
        p = read_plan(ARGS[1]).program
        p === nothing && error("plan has no evaluator")
        backend = Symbol(ARGS[4])
        trace = ARGS[5] == "ON"
        header = sprint() do io
            if ARGS[3] == "kernel"
                model = isempty(ARGS[6]) ? nothing : read_resource_model(ARGS[6])
                emit_kernel(io, p; backend, trace, resource_model = model)
            else
                emit_cpp(io, p; backend, trace, float_backend = Symbol(ARGS[2]))
            end
        end
        gmp = occursin("#include <gmpxx.h>", header)
        ac = occursin(r"#include <ac_(std_float|fixed|int)\.h>", header)
        print(gmp ? "ON" : "OFF", " ", ac ? "ON" : "OFF")
      ]=]
      "${PRESAGE_PLAN}" "${PRESAGE_FLOAT_BACKEND}" "${PRESAGE_TARGET}" "${PRESAGE_BACKEND}" "${PRESAGE_TRACE}" "${PRESAGE_RESOURCE_MODEL}"
    RESULT_VARIABLE inspect_result OUTPUT_VARIABLE resources OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT inspect_result EQUAL 0)
    message(FATAL_ERROR "Could not inspect evaluator plan dependencies")
  endif()
  separate_arguments(resources UNIX_COMMAND "${resources}")
  list(GET resources 0 needs_gmp)
  list(GET resources 1 floating_needs_ac)
  if(floating_needs_ac)
    set(needs_ac ON)
  endif()
else()
  set(source erf)
  set(options "--evaluation=${PRESAGE_EVALUATION}" "--internal=${PRESAGE_INTERNAL}")
endif()
list(APPEND options "--backend=${PRESAGE_BACKEND}")
set(resource_outputs)
if(PRESAGE_RESOURCE_MODEL)
  list(APPEND options "--resource-model=${PRESAGE_RESOURCE_MODEL}")
  set(resource_outputs "${generated}/resources.toml" "${generated}/resource-model.toml")
  if(NOT PRESAGE_PLAN AND PRESAGE_INTERNAL STREQUAL "fixed" AND PRESAGE_CERTIFY)
    list(APPEND resource_outputs "${generated}/implementations.toml")
  endif()
endif()
list(APPEND options "--target=${PRESAGE_TARGET}")
set(trace_options)
if(PRESAGE_TRACE)
  list(APPEND options --trace=true)
  set(trace_options --trace)
endif()
if(PRESAGE_TARGET STREQUAL "kernel")
  set(needs_gmp OFF)
  list(APPEND options "--toolchain=${PRESAGE_TOOLCHAIN}")
else()
  list(APPEND options "--float-backend=${PRESAGE_FLOAT_BACKEND}")
endif()
set(certificate_output)
if(PRESAGE_CERTIFY)
  list(APPEND options --certify=true)
  set(certificate_output "${generated}/certificate.toml")
else()
  list(APPEND options --certify=false)
endif()
add_custom_command(
  OUTPUT "${generated}/evaluator.hpp" "${generated}/evaluate.cpp" "${generated}/plan.toml" ${certificate_output} ${resource_outputs}
  COMMAND "${PRESAGE_JULIA}" --startup-file=no "--project=${CMAKE_CURRENT_SOURCE_DIR}"
          "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate.jl" "${source}" "${generated}" ${options}
  DEPENDS ${PRESAGE_JULIA_SOURCES} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate.jl"
    "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generation_artifacts.jl"
    "${CMAKE_CURRENT_SOURCE_DIR}/scripts/polynomial_worker.jl" ${PRESAGE_PLAN} ${PRESAGE_RESOURCE_MODEL}
  VERBATIM)
add_executable(presage_cpu "${generated}/evaluate.cpp" "${generated}/evaluator.hpp")
target_compile_features(presage_cpu PRIVATE cxx_std_20)
if(needs_gmp)
  target_include_directories(presage_cpu SYSTEM PRIVATE "${PRESAGE_GMP_INCLUDE_DIR}")
  target_link_libraries(presage_cpu PRIVATE "${PRESAGE_GMPXX}" "${PRESAGE_GMP}")
endif()
if(PRESAGE_BACKEND STREQUAL "ap_fixed")
  find_path(PRESAGE_AP_INCLUDE_DIR ap_fixed.h
    HINTS "$ENV{PRESAGE_AP_INCLUDE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../HLS_arbitrary_Precision_Types/include"
    REQUIRED)
  target_include_directories(presage_cpu SYSTEM PRIVATE "${PRESAGE_AP_INCLUDE_DIR}")
endif()
find_path(PRESAGE_AC_INCLUDE_DIR ac_std_float.h
    HINTS "$ENV{PRESAGE_AC_INCLUDE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../ac_types/include"
    )
if(PRESAGE_AC_INCLUDE_DIR)
  target_include_directories(presage_cpu SYSTEM PRIVATE "${PRESAGE_AC_INCLUDE_DIR}")
elseif(needs_ac)
  message(FATAL_ERROR "AC headers required; set PRESAGE_AC_INCLUDE_DIR")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(presage_cpu PRIVATE -ffp-contract=off -fno-fast-math -frounding-math)
else()
  message(FATAL_ERROR "The CPU verification build currently requires Clang or GCC")
endif()
if(BUILD_TESTING)
  if(PRESAGE_TARGET STREQUAL "cpu")
    add_test(NAME cpu_conformance COMMAND "${PRESAGE_JULIA}" --startup-file=no
      "--project=${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify_cpu.jl"
      "${generated}/plan.toml" "$<TARGET_FILE:presage_cpu>" ${trace_options})
    set(verification_test sampled_verification)
  else()
    set(verification_test kernel_verification)
  endif()
  add_test(NAME ${verification_test} COMMAND "${PRESAGE_JULIA}" --startup-file=no
    "--project=${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify.jl"
    "${generated}/plan.toml" "-"
    "--checker=$<TARGET_FILE:presage_check>" "--evaluator=$<TARGET_FILE:presage_cpu>")
endif()
